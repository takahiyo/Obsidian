function maskToNum(mask)
{
var num,i,j;
var zero="";
var bin="";
var maskToNumValue=0;
for(i=1;i<=4;i++)
{
zero = "";
num = getDigit(mask,i);
var length = (rvt(num)).length;
if(length < 8)
{			
for(j=1;j<=(8-length);j++)
zero = zero + '0';
}		
bin = bin + zero + rvt(num);
}
for(i=0;i<32;i++)
{
if(bin.charAt(i)==1)
maskToNumValue++;
else
break;
}
return maskToNumValue;
}
function numToMask(num)
{
var i;
var bin="";
var numToMaskValue="";
for(i=0;i<parseInt(num);i++)
{
bin=bin+'1';
}
for(i=0;i<32-parseInt(num);i++)
{
bin=bin+'0';
}
numToMaskValue = parseInt(bin.substring(0,8),2)+'.'+parseInt(bin.substring(8,16),2)+'.'+parseInt(bin.substring(16,24),2)+'.'+parseInt(bin.substring(24,32),2);
return numToMaskValue;
}
