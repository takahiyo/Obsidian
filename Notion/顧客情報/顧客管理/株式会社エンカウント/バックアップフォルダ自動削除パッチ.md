```vb.net
set ospro=\\192.168.1.250
set DCshare=%ospro%\DailyCopy_Share
@rem デイリーコピーフォルダ内から30日以上経過したフォルダを削除する
forfiles /p "%DCshare%" /D -30 /c "cmd /c IF @isdir==TRUE rmdir /S /Q @file”
```
