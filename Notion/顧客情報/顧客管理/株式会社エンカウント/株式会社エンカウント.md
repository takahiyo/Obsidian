[[バックアップフォルダ自動削除パッチ]]

> [!note]+ バックアップ及び日数経過フォルダ削除ps1バッチ
> # 変数の定義
> 
> $source = "\\192.168.1.250\WeeklyCopy_Share"
> $backup = "z:\WeeklyCopy_Share_Backup"
> $share = "\\192.168.1.250\Share"
> $dateFolder = Get-Date -Format "yyyyMMdd"
> $newFolder = Join-Path -Path $source -ChildPath $dateFolder
> $tenDaysAgo = (Get-Date).AddDays(-10)
> $thirtyDaysAgo = (Get-Date).AddDays(-30)
> 
> # 1. コピー操作
> 
> robocopy $source $backup /MIR
> 
> # 2. 本日の日付で新しいフォルダを作成
> 
> New-Item -Path $newFolder -ItemType Directory
> 
> # 3. ミラーリング操作
> 
> robocopy $share $newFolder /MIR
> 
> # 4. 10日以上経過したフォルダを削除
> 
> Get-ChildItem -Path $source -Directory | Where-Object {
> $*.CreationTime -lt $tenDaysAgo
> } | ForEach-Object {
> Remove-Item $*.FullName -Recurse -Force
> }
> 
> # 5. 30日以上経過したフォルダを削除
> 
> Get-ChildItem -Path $backup -Directory | Where-Object {
> $*.CreationTime -lt $thirtyDaysAgo
> } | ForEach-Object {
> Remove-Item $*.FullName -Recurse -Force
> }

> [!note]+ 30日経過フォルダ削除　PowerShellコード
> ```javascript
> $DCshare = "\\192.168.1.250\DailyCopy_Share"
> $limitDate = (Get-Date).AddDays(-30)
> 
> Get-ChildItem -Path $DCshare | Where-Object {
>     $_.PSIsContainer -and $_.CreationTime -lt $limitDate
> } | ForEach-Object {
>     Remove-Item $_.FullName -Recurse -Force
> }
> ```
